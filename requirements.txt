# selective scan 커널은 pip 로 여기서 걸 수 없다. 아래 순서를 그대로 따를 것.
#
# 1) CUDA 11.8 툴체인. 배포판 gcc 를 쓰면 안 된다 — CUDA 11.8 의 헤더는 glibc 2.41+
#    (Ubuntu 26.04 는 2.43)와 충돌해 cospi/sinpi/rsqrt 의 exception specification 이
#    어긋나고 .cu 컴파일이 전부 깨진다. conda 의 gcc-11 + sysroot 2.17 을 host
#    compiler 로 지정해 시스템 glibc 헤더를 피한다.
#
#      conda install -n <env> -c nvidia/label/cuda-11.8.0 cuda-toolkit
#      conda install -n <env> -c conda-forge gcc_linux-64=11.4 gxx_linux-64=11.4 \
#                                            sysroot_linux-64=2.17
#      export CUDA_HOME=<env>
#      export CC=<env>/bin/x86_64-conda-linux-gnu-gcc
#      export CXX=<env>/bin/x86_64-conda-linux-gnu-g++
#      export NVCC_PREPEND_FLAGS="-ccbin $CXX"
#      export TORCH_CUDA_ARCH_LIST="8.6"   # 측정 머신이 RTX 3070 Ti 하나뿐이다
#
# 2) causal_conv1d 1.1.0. PyPI sdist(6.6 kB)에는 csrc/ 가 없어 빌드가 불가능하다.
#    반드시 소스 트리에서 빌드한다.
#
#      git clone --depth 1 --branch v1.1.0 \
#          https://github.com/Dao-AILab/causal-conv1d.git
#      pip install --no-build-isolation ./causal-conv1d
#
# 3) mamba-1p1p1 — 정식 mamba-ssm 이 아니라 Vim 의 포크여야 한다. 정식 1.1.1 의
#    Mamba 는 bimamba_type 인자를 받지 않아 models/vim_official.py 가 죽는다.
#    스모크 테스트가 import 하는 selective_scan_fn 은 양쪽 모두에 있어서 이 차이를
#    잡아주지 못한다.
#
#      git clone --depth 1 https://github.com/hustvl/Vim.git
#      pip install --no-build-isolation -e ./Vim/mamba-1p1p1
